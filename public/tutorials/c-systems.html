Module 1 · Compilation & Architecture

The 4-Stage GCC Compilation Pipeline & Process Memory

Understand Preprocessing, Compilation, Assembly, Linking, and virtual memory segments.

1. GCC Compilation Toolchain Stages

[app.c] 
   │
   ▼ 1. Preprocessor (cpp / gcc -E): Expands #include, #define macros, strips comments
[app.i]
   │
   ▼ 2. Compiler (gcc -S): Parses AST, typechecks, generates target Assembly code
[app.s]
   │
   ▼ 3. Assembler (as / gcc -c): Translates Assembly instructions to Object Bytecode
[app.o] (Relocatable Object File)
   │
   ▼ 4. Linker (ld / gcc -o): Resolves external symbols (libc.so), sets entry points
[app] (ELF / Mach-O / PE Binary Executable)

2. Linux Process Virtual Memory Segments

SegmentAddress RangeRead/Write PermissionsContents & Lifetime
StackHigh memory $\downarrow$Read / WriteStack frames, local auto variables, return pointers. Cleared on function exit.
HeapLow memory $\uparrow$Read / WriteDynamic allocation (malloc). Persists until explicit free().
BSSFixedRead / WriteUninitialized static and global variables. Zeroed by OS kernel on startup.

Module 2 · Types & Bits

Data Types, Two's Complement, Overflow & Bitwise Math

Integer widths, sign representation, bitmasking, and bit shifts.

#include <stdio.h>
#include <stdint.h>

int main() {
    uint8_t flags = 0b00000101; // Bitmask: Bit 0 and Bit 2 set

    // Set Bit 1: flags |= (1 << 1)
    flags |= (1 << 1); // 0b00000111 (7)

    // Clear Bit 0: flags &= ~(1 << 0)
    flags &= ~(1 << 0); // 0b00000110 (6)

    // Toggle Bit 2: flags ^= (1 << 2)
    flags ^= (1 << 2); // 0b00000010 (2)

    printf("Updated flags byte: %u\n", flags);
    return 0;
}
Module 3 · Pointers

Pointers, Memory Addresses & Dereferencing Mechanics

Understand address-of (`&`), dereference operator (`*`), pointer sizing (32-bit vs 64-bit), and pass-by-reference.

#include <stdio.h>

void swapByReference(int *a, int *b) {
    int temp = *a; // Dereference pointer 'a' to read integer value
    *a = *b;       // Store value of 'b' into memory location pointed by 'a'
    *b = temp;     // Store temp into memory location pointed by 'b'
}

int main() {
    int x = 10, y = 20;
    printf("Before: x=%d, y=%d\n", x, y);
    swapByReference(&x, &y); // Pass physical memory addresses
    printf("After:  x=%d, y=%d\n", x, y); // x=20, y=10
    return 0;
}
Module 4 · Pointer Arithmetic

Pointer Arithmetic, Arrays & Memory Offsets

Why `arr[i]` is identical to `*(arr + i)` and how pointer scaling works with `sizeof(T)`.

#include <stdio.h>

int main() {
    int nums[] = {100, 200, 300, 400};
    int *ptr = nums; // Points to nums[0]

    // Pointer arithmetic advances by sizeof(int) = 4 bytes!
    printf("ptr = %p, *ptr = %d\n", (void*)ptr, *ptr);         // 100
    printf("ptr+1 = %p, *(ptr+1) = %d\n", (void*)(ptr+1), *(ptr+1)); // 200

    return 0;
}
Module 5 · Strings & Security

C Strings, Null Terminator (`\0`) & Buffer Overflow Vulnerabilities

Dangers of unbounded copy functions (`gets`, `strcpy`, `strcat`) and modern safe alternatives (`snprintf`, `strncat`).

#include <stdio.h>
#include <string.h>

int main() {
    char dest[16];
    const char *src = "Secure Systems Engineering";

    // VULNERABLE: strcpy(dest, src); // Overwrites return address on stack!
    // SAFE ALTERNATIVE:
    snprintf(dest, sizeof(dest), "%s", src);
    printf("Safely bounded buffer: '%s' (Length: %lu)\n", dest, strlen(dest));
    return 0;
}
Module 6 · Dynamic Allocation

Dynamic Heap Memory: `malloc`, `calloc`, `realloc` & `free`

Heap allocation lifetimes, zero-initialized buffers, and resizing dynamic arrays.

#include <stdio.h>
#include <stdlib.h>

int main() {
    // Allocate array of 5 integers initialized to zero
    int *scores = (int*)calloc(5, sizeof(int));
    if (!scores) { perror("Allocation failed"); return 1; }

    for (int i = 0; i < 5; i++) scores[i] = (i + 1) * 20;

    // Resize array to 10 integers
    int *resized = (int*)realloc(scores, 10 * sizeof(int));
    if (!resized) { free(scores); return 1; }
    scores = resized;

    // Always free heap memory when finished
    free(scores);
    scores = NULL; // Defend against dangling pointer
    return 0;
}
Module 7 · Memory Debugging

Memory Leaks, Dangling Pointers, Double Free & Valgrind Diagnostics

Detect and eliminate heap memory leaks using `valgrind --leak-check=full`.

⚠️ The 3 Classic C Memory Bugs
  • Memory Leak: Allocating on heap without corresponding free() before losing pointer.
  • Dangling Pointer: Accessing memory address after it has already been freed.
  • Double Free: Calling free() twice on the same pointer, corrupting the heap allocator metadata.

Module 9 · Function Pointers

Function Pointers, Callbacks & Polymorphism in C

Store executable function addresses and implement custom sorting comparators with `qsort`.

#include <stdio.h>
#include <stdlib.h>

int compareDescending(const void *a, const void *b) {
    int valA = *(const int*)a;
    int valB = *(const int*)b;
    return valB - valA; // Positive if B > A (Descending)
}

int main() {
    int arr[] = {42, 12, 88, 3, 99, 56};
    qsort(arr, 6, sizeof(int), compareDescending);
    
    printf("Sorted Descending: ");
    for (int i = 0; i < 6; i++) printf("%d ", arr[i]);
    printf("\n");
    return 0;
}
Module 10 · File I/O

File Streams, Binary Serialization (`fread`/`fwrite`) & Buffering

Persist and deserialize structs directly to binary disk streams.

#include <stdio.h>

typedef struct {
    int id;
    char name[32];
    float gpa;
} Student;

int main() {
    Student s = {101, "Aarav Sharma", 3.85f};

Module 13 · Process Control

Process Creation & Lifecycle: `fork()`, `execvp()` & `waitpid()`

Clone address spaces with Copy-On-Write (COW), spawn subprocesses, and prevent Zombie processes.

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    pid_t pid = fork();

    if (pid == 0) {
        // Child Process
        printf("[CHILD] Executing in PID %d\n", getpid());
        execlp("ls", "ls", "-l", NULL);
    } else if (pid > 0) {
        // Parent Process
        waitpid(pid, NULL, 0); // Reaps child, preventing Zombie state
        printf("[PARENT] Child finished.\n");
    }
    return 0;
}
Module 14 · Concurrency

POSIX Multithreading (`pthreads`) & Mutex Synchronization

Shared memory concurrency, race conditions, and critical section protection via `pthread_mutex_t`.

#include <stdio.h>
#include <pthread.h>

int counter = 0;
pthread_mutex_t lock;

void* worker(void* arg) {
    for (int i = 0; i < 10000; i++) {
        pthread_mutex_lock(&lock);   // Enter Critical Section
        counter++;
        pthread_mutex_unlock(&lock); // Leave Critical Section
    }
    return NULL;
}

int main() {
    pthread_t t1, t2;
    pthread_mutex_init(&lock, NULL);

    pthread_create(&t1, NULL, worker, NULL);
    pthread_create(&t2, NULL, worker, NULL);

    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    pthread_mutex_destroy(&lock);
    printf("Final synchronized counter: %d\n", counter); // Guaranteed 20000
    return 0;
}
